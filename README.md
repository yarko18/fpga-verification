## install
    git clone git@gitlab.com:fpga8683920/cocotb_helpers.git
    cd cocotb_helpers
    python.exe -m pip install -e .

## reinstall 

    python.exe -m pip uninstall fpga-cocotb -y
    python.exe -m pip install -e .


## check installation

    python.exe -m pip show fpga-cocotb
